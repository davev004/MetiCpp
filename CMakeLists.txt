cmake_minimum_required(VERSION 3.20)
project(MetiC++ VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(METI_NATIVE "Enable host-specific CPU instructions" OFF)
option(METI_STATIC "Statically link the GCC runtime libraries" OFF)
set(METI_ARCH "generic" CACHE STRING "Target instruction set: generic, popcnt, avx2, or avx512")
set_property(CACHE METI_ARCH PROPERTY STRINGS generic popcnt avx2 avx512)
set(_METI_VALID_ARCHES generic popcnt avx2 avx512)
if(NOT METI_ARCH IN_LIST _METI_VALID_ARCHES)
	message(FATAL_ERROR "METI_ARCH must be one of: generic, popcnt, avx2, avx512")
endif()

# Core source files
file(GLOB_RECURSE SRC_FILES "src/*.cpp")

# Target 1: Standard Modern CPU (AVX2 / x86-64-v3)
add_executable(meticpp ${SRC_FILES})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
	target_compile_options(meticpp PRIVATE -Wall -Wextra)
	target_compile_options(meticpp PRIVATE $<$<CONFIG:Release>:-O3;-flto>)

	if(METI_NATIVE)
		target_compile_options(meticpp PRIVATE -march=native)
	elseif(METI_ARCH STREQUAL "popcnt")
		target_compile_options(meticpp PRIVATE -march=x86-64 -mpopcnt)
	elseif(METI_ARCH STREQUAL "avx2")
		target_compile_options(meticpp PRIVATE -march=x86-64-v3)
	elseif(METI_ARCH STREQUAL "avx512")
		target_compile_options(meticpp PRIVATE -march=x86-64-v4)
	endif()

	if(METI_STATIC)
		target_link_options(meticpp PRIVATE -static-libgcc -static-libstdc++)
		if(WIN32)
			target_link_options(meticpp PRIVATE -static)
		endif()
	endif()
endif()
find_package(Threads REQUIRED)
target_link_libraries(meticpp PRIVATE Threads::Threads)

enable_testing()
add_executable(meticpp_tests tests/perft.cpp src/move_logic.cpp)
target_include_directories(meticpp_tests PRIVATE src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
	target_compile_options(meticpp_tests PRIVATE -Wall -Wextra)
	target_compile_options(meticpp_tests PRIVATE $<$<CONFIG:Release>:-O3;-flto>)
endif()
target_link_libraries(meticpp_tests PRIVATE Threads::Threads)
add_test(NAME meticpp_perft COMMAND meticpp_tests)
